cmake_minimum_required(VERSION 3.19)
project(warehouse_tapf_experiment CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LMRP_SOURCE_DIR "" CACHE PATH "Unmodified pinned libMultiRobotPlanning checkout")
set(expected "4c75fa20c435c440d8b6bd6dc81668ddc7296ba0")
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${LMRP_SOURCE_DIR}"
  OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT actual STREQUAL expected)
  message(FATAL_ERROR "libMultiRobotPlanning must match ${expected}")
endif()
execute_process(COMMAND git diff --exit-code HEAD WORKING_DIRECTORY "${LMRP_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
find_package(Boost REQUIRED COMPONENTS program_options)
find_package(yaml-cpp REQUIRED)
foreach(program IN ITEMS assignment cbs cbs_ta)
  add_executable(${program} "${LMRP_SOURCE_DIR}/example/${program}.cpp")
  target_include_directories(${program} PRIVATE "${LMRP_SOURCE_DIR}/include")
  target_link_libraries(${program} PRIVATE Boost::program_options yaml-cpp::yaml-cpp)
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/provenance.json"
  "{\"url\":\"https://github.com/whoenig/libMultiRobotPlanning\",\"commit\":\"${actual}\",\"license\":\"MIT\",\"source\":\"${LMRP_SOURCE_DIR}\"}\n")
