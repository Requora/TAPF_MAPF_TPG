cmake_minimum_required(VERSION 3.19)
project(warehouse_native_bridges CXX)
set(CMAKE_CXX_STANDARD 20)
set(MAWPF_SOURCE_DIR "" CACHE PATH "Pinned MAWPF checkout")
set(WINKTPG_SOURCE_DIR "" CACHE PATH "Pinned WinkTPG checkout")
foreach(pair IN ITEMS "MAWPF;0cdb8e43797aff3781e4cd3ddb2c6d7d36f3df17" "WINKTPG;dd96946f225e206de35b041320786eab43210658")
  list(GET pair 0 upstream)
  list(GET pair 1 expected)
  execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${${upstream}_SOURCE_DIR}" OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "${upstream}: checkout must match ${expected}")
  endif()
  execute_process(COMMAND git diff --exit-code HEAD WORKING_DIRECTORY "${${upstream}_SOURCE_DIR}" COMMAND_ERROR_IS_FATAL ANY)
endforeach()
add_subdirectory("${MAWPF_SOURCE_DIR}/algorithms" mawpf-algorithms)
add_executable(warehouse_mawpf mawpf.cpp)
target_link_libraries(warehouse_mawpf PRIVATE algorithms)
find_package(Boost REQUIRED COMPONENTS system program_options filesystem)
file(GLOB WINKTPG_SOURCES "${WINKTPG_SOURCE_DIR}/src/*.cpp")
list(REMOVE_ITEM WINKTPG_SOURCES "${WINKTPG_SOURCE_DIR}/src/driver.cpp")
add_executable(warehouse_winktpg winktpg.cpp ${WINKTPG_SOURCES})
target_include_directories(warehouse_winktpg PRIVATE "${WINKTPG_SOURCE_DIR}/inc" ${Boost_INCLUDE_DIRS})
target_compile_options(warehouse_winktpg PRIVATE -include unordered_map)
target_link_libraries(warehouse_winktpg PRIVATE ${Boost_LIBRARIES})
